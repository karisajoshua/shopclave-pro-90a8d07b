## Dashboard metric definitions (after fix)
All in CAD; window = `orders.created_at > cutoff`.
- Paid orders: count `orders` with `payment_status='paid'`. Hint shows total / awaiting payment / cancelled.
- Net paid sales: Σ paid `orders.total` − Σ `order_items.refunded_amount` of those orders.
- Platform earnings: Σ `order_items.commission_amount` of paid orders.
- Vendor payouts: Σ completed `withdrawal_requests.amount` with processed_at (or requested_at) in window.
- Top vendors/products & chart: paid orders only. Products: exact count query.
- All lists paginate via `fetchAllRows`; any failure shows a warning banner instead of zeros.

## Proposed migrations (NOT applied)
```sql
-- M1 transition guard + status model
CREATE OR REPLACE FUNCTION public.guard_item_status_transition() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE _pay text; _flow text[] := ARRAY['pending','processing','shipped','delivered'];
BEGIN
  IF NEW.status IS NOT DISTINCT FROM OLD.status THEN RETURN NEW; END IF;
  IF auth.uid() IS NULL OR public.has_role(auth.uid(),'admin') THEN RETURN NEW; END IF;
  SELECT payment_status INTO _pay FROM public.orders WHERE id = NEW.order_id;
  IF _pay <> 'paid' THEN RAISE EXCEPTION 'Order is not paid yet'; END IF;
  IF array_position(_flow, NEW.status) IS DISTINCT FROM array_position(_flow, OLD.status) + 1 THEN
    RAISE EXCEPTION 'Invalid status change % -> %', OLD.status, NEW.status;
  END IF;
  RETURN NEW;
END $$;
CREATE TRIGGER trg_guard_item_status BEFORE UPDATE OF status ON public.order_items
  FOR EACH ROW EXECUTE FUNCTION public.guard_item_status_transition();
-- sync_order_status_from_items: exclude cancelled lines; all cancelled -> 'cancelled'.

-- M2 notification on status change (replaces client insert)
CREATE OR REPLACE FUNCTION public.notify_item_status() RETURNS trigger
LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
  IF NEW.status IS DISTINCT FROM OLD.status THEN
    INSERT INTO public.notifications(recipient_id, title, message, type)
    SELECT o.user_id, 'Order Update', 'An item in your order is now ' || NEW.status, 'order'
    FROM public.orders o WHERE o.id = NEW.order_id AND o.user_id IS NOT NULL;
  END IF; RETURN NEW;
END $$;
CREATE TRIGGER trg_notify_item_status AFTER UPDATE OF status ON public.order_items
  FOR EACH ROW EXECUTE FUNCTION public.notify_item_status();

-- M3 realtime
ALTER PUBLICATION supabase_realtime ADD TABLE public.orders, public.order_items,
  public.shipments, public.tracking_events, public.notifications;

-- M4 indexes
CREATE INDEX IF NOT EXISTS idx_order_items_order ON public.order_items(order_id);
CREATE INDEX IF NOT EXISTS idx_order_items_vendor ON public.order_items(vendor_id);
CREATE INDEX IF NOT EXISTS idx_notifications_recipient ON public.notifications(recipient_id, created_at DESC);
```
